cmake_minimum_required(VERSION 3.20)
project(CosmicSynapseLivingUniverse VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(COSMOS_BUILD_GAME "Build the raylib game" ON)
option(COSMOS_BUILD_TESTS "Build deterministic headless smoke test" ON)
add_library(cosmos_core cpp/src/universe.cpp cpp/src/cst_state.cpp cpp/src/live_data.cpp cpp/src/persistence.cpp)
target_include_directories(cosmos_core PUBLIC cpp/include)
if(COSMOS_BUILD_TESTS)
 add_executable(cosmos_headless cpp/src/headless_test.cpp)
 target_link_libraries(cosmos_headless PRIVATE cosmos_core)
endif()
if(COSMOS_BUILD_GAME)
 include(FetchContent)
 set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 6.0 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(raylib)
 add_executable(cosmos_universe cpp/src/main.cpp cpp/src/renderer.cpp)
 target_link_libraries(cosmos_universe PRIVATE cosmos_core raylib)
 target_include_directories(cosmos_universe PRIVATE cpp/include)
 if(APPLE)
  target_link_libraries(cosmos_universe PRIVATE "-framework IOKit" "-framework Cocoa" "-framework OpenGL")
 endif()
endif()
